\newglossaryentry{mssql}{
    name=MSSQL,
    description={Microsoft SQL Server, ein relationales Datenbankmanagementsystem
    zur Speicherung, Verwaltung und Abfrage strukturierter Daten}
}

\newglossaryentry{grafana}{
    name=Grafana,
    description={Open-Source-Software zur Visualisierung von Überwachungsdaten, 
    die es ermöglicht, Daten aus unterschiedlichen Quellen mithilfe von 
    Dashboards und Plugins grafisch darzustellen und auszuwerten}
}

\newglossaryentry{silverpeak}{
    name=Silverpeak Appliance,
    description={Netzwerkgerät des Herstellers Silverpeak zur Bereitstellung
    und Überwachung von \gls{sdwan}-Verbindungen in verteilten Netzwerken}
}

\newglossaryentry{sdwan}{
    name=SD-WAN,
    description={Software-Defined Wide Area Network, eine softwarebasierte
    Technologie zur zentralen Steuerung und Verwaltung von
    WAN-Verbindungen über verschiedene Netzwerkverbindungen}
}

\newglossaryentry{wan}{
    name=WAN,
    description={Wide Area Network, ein Weitverkehrsnetzwerk zur Verbindung
    räumlich voneinander getrennter Netzwerke und Standorte}
}

\newglossaryentry{hci}{
    name=HCI,
    description={Hyperconverged Infrastructure, eine softwaredefinierte
    Infrastruktur, bei der Rechenleistung, Speicher und Netzwerkfunktionen
    in einem einheitlichen System zusammengeführt und zentral verwaltet
    werden~\cite{redhat2026hci}}
}
\newglossaryentry{san}{
    name=SAN,
    description={Storage Area Network, ein dediziertes Speichernetzwerk, das
    mehrere Speichergeräte zu einer zentral verwalteten, gemeinsam nutzbaren
    Ressource zusammenfasst und Servern einen schnellen, blockbasierten
    Zugriff auf Daten ermöglicht~\cite{ibm2026san}}
}
\newglossaryentry{vsan}{
    name=vSAN,
    description={Virtual SAN, eine softwaredefinierte Speicherlösung von
    VMware, die lokalen Speicher mehrerer Server zu einem gemeinsamen,
    zentral verwalteten Datenspeicher zusammenfasst~\cite{vmware2026vsan}}
}
\newglossaryentry{raid}{ 
    name=RAID, 
    description={Redundant Array of Independent Disks, ein Verfahren zur 
    Zusammenfassung mehrerer physischer Datenträger zu einer logischen 
    Speichereinheit, um die Leistung, Datenverfügbarkeit und Ausfallsicherheit 
    von Speichersystemen zu verbessern~\cite{ibm2026raid}}
}
\newglossaryentry{zabbix}{
    name=Zabbix,
    description={Open-Source-Monitoring-Software zur Überwachung von Servern,
    Netzwerkgeräten und weiteren Systemen, die eine Erfassung und
    Auswertung von Zustands- und Leistungsdaten sowie eine Gruppierung
    der überwachten Systeme über Tags und Gruppen ermöglicht
    ~\cite{zabbix2026about}}
}
\newglossaryentry{jsonrpc}{
    name=JSON-RPC,
    description={Zustandsloses, leichtgewichtiges Remote-Procedure-Call-Protokoll, 
    das \gls{json} als Datenformat verwendet und transportunabhängig eingesetzt 
    werden kann, beispielsweise über HTTP oder Sockets~\cite{jsonrpc2013spec}}
}
\newglossaryentry{edgeconnectorchestrator}{ 
    name=EdgeConnect Orchestrator, 
    description={Zentrale Management- und Orchestrierungsplattform von Aruba EdgeConnect 
    zur Konfiguration, Überwachung und Verwaltung von \gls{sdwan}-Infrastrukturen. Die Plattform

    über eine einheitliche Verwaltungsoberfläche ~\cite{aruba2026userguide}} 
}
\newglossaryentry{rest}{
    name=REST,
    description={Representational State Transfer, ein Architekturstil für
    verteilte Systeme, der durch einheitliche Schnittstellen,
    Ressourcenorientierung und zustandslose Kommunikation gekennzeichnet
    ist~\cite{fielding2000rest}}
}
\newglossaryentry{sessioncookie}{
    name=Session-Cookie,
    description={Cookie, das von einem Webserver zur Zuordnung eines
    Benutzers zu einer aktiven Sitzung verwendet wird. Es wird vom
    Client bei nachfolgenden HTTP-Anfragen erneut an den Server
    übertragen und ermöglicht dadurch die Aufrechterhaltung einer
    Sitzung über mehrere Anfragen hinweg~\cite{rfc6265}}
}
\newglossaryentry{silverpeakcompany}{
    name=Silverpeak,
    description={Hersteller von SD-WAN-Lösungen, dessen Produkte über die
    Plattform \gls{edgeconnectorchestrator} zentral verwaltet und
    überwacht werden können~\cite{aruba2026userguide}}
}
\newglossaryentry{csrftoken}{
    name=CSRF-Token,
    description={Zufällig generierter, sitzungsgebundener Wert, der zum
    Schutz vor Cross-Site-Request-Forgery-Angriffen eingesetzt wird. Der
    Server übermittelt den Token sowohl als Cookie als auch erwartet ihn
    bei zustandsändernden Anfragen zusätzlich als HTTP-Header oder
    Formularfeld; stimmen beide Werte nicht überein, wird die Anfrage
    abgelehnt~\cite{owasp2026csrf}}
}
