\newglossaryentry{mssql}{
    name=MSSQL,
    description={Microsoft SQL Server, a relational database management system
    used for storing, managing, and retrieving structured data}
}

\newglossaryentry{grafana}{
    name=Grafana,
    description={Grafana, an open-source monitoring and visualization software 
    solution that allows data from different sources to be displayed using
    dashboards and plugins}
}

\newglossaryentry{silverpeak}{
    name=Silver Peak Appliance,
    description={Netzwerkgerät des Herstellers Silver Peak zur Bereitstellung
    und Überwachung von \gls{sdwan}-Verbindungen in verteilten Netzwerken}
}

\newglossaryentry{sdwan}{
    name=SD-WAN,
    description={Software-Defined Wide Area Network, eine softwarebasierte
    Technologie zur zentralen Steuerung und Verwaltung von
    WAN-Verbindungen über verschiedene Netzwerkverbindungen}
}

\newglossaryentry{wan}{
    name=WAN,
    description={Wide Area Network, ein Weitverkehrsnetzwerk zur Verbindung
    räumlich voneinander getrennter Netzwerke und Standorte}
}

\newglossaryentry{hci}{
    name=HCI,
    description={Hyperconverged Infrastructure, eine softwaredefinierte
    Infrastruktur, bei der Rechenleistung, Speicher und Netzwerkfunktionen
    in einem einheitlichen System zusammengeführt und zentral verwaltet
    werden~\cite{redhat2026hci}}
}
\newglossaryentry{san}{
    name=SAN,
    description={Storage Area Network, ein dediziertes Speichernetzwerk, das
    mehrere Speichergeräte zu einer zentral verwalteten, gemeinsam nutzbaren
    Ressource zusammenfasst und Servern einen schnellen, blockbasierten
    Zugriff auf Daten ermöglicht~\cite{ibm2026san}}
}
\newglossaryentry{vsan}{
    name=vSAN,
    description={Virtual SAN, eine softwaredefinierte Speicherlösung von
    VMware, die lokalen Speicher mehrerer Server zu einem gemeinsamen,
    zentral verwalteten Datenspeicher zusammenfasst~\cite{vmware2026vsan}}
}
\newglossaryentry{raid}{ 
    name=RAID, 
    description={Redundant Array of Independent Disks, ein Verfahren zur 
    Zusammenfassung mehrerer physischer Datenträger zu einer logischen 
    Speichereinheit, um die Leistung, Datenverfügbarkeit und Ausfallsicherheit 
    von Speichersystemen zu verbessern~\cite{ibm2026raid}}
}
